public class Solution {
    public string FindLexSmallestString(string s, int a, int b) {
        var visited = new HashSet<string>();
        var q = new Queue<string>();
        q.Enqueue(s);
        visited.Add(s);

        string smallest = s;
        while(q.Count>0){
            string current = q.Dequeue();
            if(string.Compare(current, smallest) < 0) smallest = current;

            char[] chars = current.ToCharArray();
            for(int i=1; i<chars.Length; i+=2) chars[i] = (char)(((chars[i]-'0'+a)%10)+'0');

            string added = new string(chars);
            if(visited.Add(added)) q.Enqueue(added);

            string rotated = current.Substring(current.Length-b) + current.Substring(0,current.Length-b);
            if (visited.Add(rotated)) q.Enqueue(rotated);
        }
        return smallest;
    }
}
